```python
class PositionSizingPolicy(BaseModel):
    max_risk_per_trade: float = 0.01  # 1% max risk per trade
    max_position_size: float = 0.10  # 10% max in any single position
    min_position_size: float = 0.001  # 0.1% min to avoid dust
    sizing_method: Literal["fixed", "kelly", "percent_risk"] = "percent_risk"

class DrawdownProtectionPolicy(BaseModel):
    max_portfolio_drawdown: float = 0.15  # 15% max drawdown
    max_daily_loss: float = 0.05  # 5% max daily loss
    trailing_stop_enabled: bool = True
    trailing_stop_distance: float = 0.03  # 3% trailing stop
    kill_switch_drawdown: float = 0.20  # 20% triggers kill-switch

class RegimeRiskPolicy(BaseModel):
    trending_max_exposure: float = 0.20
    ranging_max_exposure: float = 0.15
    high_volatility_max_exposure: float = 0.10
    low_volatility_max_exposure: float = 0.25
    regime_confidence_threshold: float = 0.60

class StrategyRiskPolicy(BaseModel):
    enabled_strategies: list[str] = Field(default_factory=list)  # Empty = all enabled
    blacklisted_strategies: list[str] = Field(default_factory=list)
    max_strategy_allocation: dict[str, float] = Field(default_factory=dict)
    strategy_min_sharpe: float = 0.5  # Min Sharpe to allow strategy

class PortfolioRiskPolicy(BaseModel):
    max_total_exposure: float = 0.50  # 50% max total exposure
    max_correlation: float = 0.70  # Max correlation between positions
    max_concentration: float = 0.30  # Max 30% in any one symbol
    max_active_positions: int = 10

class EmergencyPolicy(BaseModel):
    kill_switch_enabled: bool = True
    kill_switch_conditions: list[str] = Field(default_factory=lambda: [
        "portfolio_drawdown_exceeded",
        "daily_loss_exceeded",
        "consecutive_losses_10",
        "manual_trigger"
    ])
    cooldown_period_hours: int = 24
    require_manual_reset: bool = True

class RiskPolicy(BaseModel):
    position_sizing: PositionSizingPolicy = Field(default_factory=PositionSizingPolicy)
    drawdown: DrawdownProtectionPolicy = Field(default_factory=DrawdownProtectionPolicy)
    regime: RegimeRiskPolicy = Field(default_factory=RegimeRiskPolicy)
    strategy: StrategyRiskPolicy = Field(default_factory=StrategyRiskPolicy)
    portfolio: PortfolioRiskPolicy = Field(default_factory=PortfolioRiskPolicy)
    emergency: EmergencyPolicy = Field(default_factory=EmergencyPolicy)
```

### 3. Risk Check Results Model (`src/baet/risk/checks.py`)
**Purpose**: Define the output structure for risk check decisions.

**Models**:
```python
class RiskCheckResult(BaseModel):
    approved: bool
    action: str  # Original action (BUY, SELL, HOLD)
    adjusted_action: str  # May be modified to HOLD or reduced size
    adjusted_size: float | None  # May be reduced
    confidence: float  # Confidence in the decision
    reasons: list[str]  # Why approved/rejected
    risk_score: float  # 0.0 (safe) to 1.0 (high risk)
    metadata: dict[str, object]  # Additional context

class RiskViolation(BaseModel):
    rule: str  # Which rule was violated
    severity: Literal["warning", "critical", "kill_switch"]
    message: str
    current_value: float | None
    threshold_value: float | None
```

### 4. Centralized Risk Engine (`src/baet/risk/engine.py`)
**Purpose**: The core risk engine that ALL trade signals must pass through.

**Key functions**:
- `evaluate_signal(signal: dict, portfolio_state: dict, regime: RegimeLabel | None) -> RiskCheckResult`
  - Runs all pre-trade checks
  - Returns approved/rejected with reasons
  - Can adjust position size or convert to HOLD

- `check_position_sizing(signal, portfolio_state) -> RiskCheckResult`
  - Validate max position size
  - Validate max risk per trade
  - Apply sizing method (fixed, kelly, percent_risk)

- `check_drawdown(signal, portfolio_state) -> RiskCheckResult`
  - Check current drawdown vs limits
  - Check daily loss limits
  - Evaluate trailing stop conditions

- `check_regime_risk(signal, regime) -> RiskCheckResult`
  - Adjust allowed exposure based on regime
  - High volatility = reduce size
  - Low volatility = may increase size

- `check_strategy_risk(signal, strategy_name) -> RiskCheckResult`
  - Is strategy blacklisted?
  - Does strategy meet minimum Sharpe?
  - Strategy-specific allocation limits

- `check_portfolio_risk(signal, portfolio_state) -> RiskCheckResult`
  - Total exposure check
  - Concentration check
  - Correlation check (if multiple positions)

- `check_emergency_conditions(portfolio_state) -> RiskCheckResult`
  - Kill-switch evaluation
  - Emergency stop conditions
  - Returns critical if kill-switch triggered
