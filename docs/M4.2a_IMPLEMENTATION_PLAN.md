#### `src/baet/paper/portfolio.py` - Portfolio State
**Purpose**: Track paper portfolio state (cash, positions, equity).

**Class**: `PaperPortfolio`
```python
class PaperPortfolio:
    """Paper trading portfolio state."""

    def __init__(self, initial_balance: float = 10000.0):
        self.initial_balance = initial_balance
        self.cash = initial_balance
        self.positions: dict[str, dict] = {}  # symbol -> {units, avg_price}
        self.trades: list[dict] = []
        self.equity_curve: list[dict] = []

    def buy(self, symbol: str, units: float, price: float, fee: float) -> bool:
        """Execute paper buy order."""
        cost = units * price + fee
        if cost > self.cash:
            return False  # Insufficient cash

        self.cash -= cost

        if symbol not in self.positions:
            self.positions[symbol] = {'units': 0, 'avg_price': 0.0}

        # Update position
        pos = self.positions[symbol]
        total_units = pos['units'] + units
        pos['avg_price'] = ((pos['units'] * pos['avg_price']) + (units * price)) / total_units
        pos['units'] = total_units

        self.trades.append({
            'timestamp': datetime.now(),
            'symbol': symbol,
            'side': 'BUY',
            'units': units,
            'price': price,
            'fee': fee,
        })
        return True

    def sell(self, symbol: str, units: float, price: float, fee: float) -> bool:
        """Execute paper sell order."""
        if symbol not in self.positions:
            return False
        if self.positions[symbol]['units'] < units:
            return False  # Insufficient units

        proceeds = units * price - fee
        self.cash += proceeds

        pos = self.positions[symbol]
        pos['units'] -= units
        if pos['units'] == 0:
            del self.positions[symbol]

        self.trades.append({
            'timestamp': datetime.now(),
            'symbol': symbol,
            'side': 'SELL',
            'units': units,
            'price': price,
            'fee': fee,
        })
        return True

    def get_total_value(self, current_prices: dict[str, float] | None = None) -> float:
        """Calculate total portfolio value."""
        value = self.cash
        for symbol, pos in self.positions.items():
            price = current_prices.get(symbol, pos['avg_price']) if current_prices else pos['avg_price']
            value += pos['units'] * price
        return value

    def get_positions(self) -> dict:
        """Get current positions."""
        return self.positions.copy()
```

**Purpose**: Simulate order execution with slippage and fees.

**Class**: `PaperOrderSimulator`
```python
class PaperOrderSimulator:
    """Simulates order execution with realistic assumptions."""

    def __init__(
        self,
        fee_rate: float = 0.001,
        slippage_rate: float = 0.0005,
    ):
        self.fee_rate = fee_rate
        self.slippage_rate = slippage_rate

    def simulate_buy(
        self,
        price: float,
        units: float,
        side: str = "BUY",
    ) -> tuple[float, float, float]:
        """
        Simulate a buy order.
        Returns: (fill_price, units_received, fee)
        """
        fill_price = price * (1 + self.slippage_rate)
        fee = units * fill_price * self.fee_rate
        return fill_price, units, fee

    def simulate_sell(
        self,
        price: float,
        units: float,
        side: str = "SELL",
    ) -> tuple[float, float, float]:
        """
        Simulate a sell order.
        Returns: (fill_price, proceeds, fee)
        """
        fill_price = price * (1 - self.slippage_rate)
        gross = units * fill_price
        fee = gross * self.fee_rate
        return fill_price, gross - fee, fee
```
